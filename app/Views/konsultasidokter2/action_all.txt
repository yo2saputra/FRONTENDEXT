SELECT a.*,b.fullname,b.gender,b.birth_dt,b.addr,b.mobile_no,c.nama_dokter,d.SubUnit_Name as tujuan_registrasi_desc,e.fld_desc as gender_desc,f.asr_nm as asr_nm,no_kwitansi FROM tregistrasi a                               
   LEFT JOIN tmst_patient b ON a.no_pasien = b.patient_no                               
   LEFT JOIN tf_list_namadokter() c ON a.dokter = c.emp_cd                            
   --LEFT JOIN (SELECT * FROM tmst_polyclinic) d ON a.tujuan_registrasi = d.poly_cd  
   LEFT JOIN (SELECT SubUnit_Name FROM MI_Sub_Unit) d ON a.tujuan_registrasi = d.SubUnit_ID  
   LEFT JOIN tf_tfieldvalue('gender') e ON b.gender = e.fld_valu                              
   LEFT JOIN (SELECT asr_nm FROM tasuransi) f ON a.asr_cd = f.asr_cd                    
   LEFT JOIN (SELECT no_kwitansi,no_registrasi FROM tbilling) g ON a.no_registrasi = g.no_registrasi                      
   WHERE tgl_praktek = CONVERT (date, GETDATE()) AND hadir = 1 AND dokter = 'E0028' AND status = 'PM'
